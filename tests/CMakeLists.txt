function(gitq_test name)
  add_executable(${name} ${name}.cpp TestRepo.h)
  target_link_libraries(${name} PRIVATE gitq_ui Qt6::Test)
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endfunction()
